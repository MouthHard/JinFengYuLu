##### 4.2 实现图片懒加载组件

**文件**: `src/components/LazyImage.vue`

```vue
<template>
	<div class="lazy-image" ref="container">
		<img
			v-if="isVisible"
			:src="src"
			:alt="alt"
			:class="{ loaded: isLoaded }"
			@load="onLoad"
			@error="onError"
		/>
		<div v-else class="placeholder">
			<slot name="placeholder">
				<div class="default-placeholder">加载中...</div>
			</slot>
		</div>
	</div>
</template>

<script setup lang="ts">
	import { ref, onMounted, onUnmounted } from "vue";

	interface Props {
		src: string;
		alt?: string;
		threshold?: number;
	}

	const props = withDefaults(defineProps<Props>(), {
		alt: "",
		threshold: 0.1,
	});

	const isVisible = ref(false);
	const isLoaded = ref(false);
	const container = ref<HTMLElement>();

	let observer: IntersectionObserver | null = null;

	onMounted(() => {
		if ("IntersectionObserver" in window && container.value) {
			observer = new IntersectionObserver(
				(entries) => {
					entries.forEach((entry) => {
						if (entry.isIntersecting) {
							isVisible.value = true;
							observer?.disconnect();
						}
					});
				},
				{ threshold: props.threshold },
			);
			observer.observe(container.value);
		} else {
			isVisible.value = true;
		}
	});

	onUnmounted(() => {
		observer?.disconnect();
	});

	const onLoad = () => {
		isLoaded.value = true;
	};

	const onError = () => {
		console.error("图片加载失败:", props.src);
	};
</script>

<style scoped>
	.lazy-image {
		min-height: 200px;
		position: relative;
	}

	.placeholder {
		display: flex;
		align-items: center;
		justify-content: center;
		height: 100%;
		background: #f0f0f0;
	}

	.default-placeholder {
		color: #999;
	}
</style>
```

```typescript
export const scale = (
	num: number,
	in_min: number,
	in_max: number,
	out_min: number,
	out_max: number,
): number => {
	return ((num - in_min) * (out_max - out_min)) / (in_max - in_min) + out_min;
};

export const debounce = <T extends (...args: any[]) => any>(
	func: T,
	wait: number,
): ((...args: Parameters<T>) => void) => {
	let timeout: number | null = null;

	return (...args: Parameters<T>) => {
		if (timeout) {
			clearTimeout(timeout);
		}
		timeout = window.setTimeout(() => {
			func(...args);
		}, wait);
	};
};

export const throttle = <T extends (...args: any[]) => any>(
	func: T,
	limit: number,
): ((...args: Parameters<T>) => void) => {
	let inThrottle: boolean;

	return (...args: Parameters<T>) => {
		if (!inThrottle) {
			func(...args);
			inThrottle = true;
			setTimeout(() => (inThrottle = false), limit);
		}
	};
};
```

##### 6.3 创建历史模块 Store

**文件**: `src/stores/history.ts`

```typescript
import { defineStore } from "pinia";
import { ref } from "vue";

export const useHistoryStore = defineStore("history", () => {
	const dynasties = ref<any[]>([]);
	const events = ref<any[]>([]);
	const figures = ref<any[]>([]);
	const heritage = ref<any[]>([]);
	const museums = ref<any[]>([]);
	const loading = ref(false);
	const error = ref<string | null>(null);

	const loadDynasties = async () => {
		loading.value = true;
		error.value = null;
		try {
			const data = await import("@/pages/History/data/dynasties.ts");
			dynasties.value = data.default || [];
		} catch (err) {
			error.value = "加载朝代数据失败";
			console.error("Failed to load dynasties:", err);
		} finally {
			loading.value = false;
		}
	};

	const loadEvents = async () => {
		loading.value = true;
		error.value = null;
		try {
			const data = await import("@/pages/History/data/events.ts");
			events.value = data.default || [];
		} catch (err) {
			error.value = "加载事件数据失败";
			console.error("Failed to load events:", err);
		} finally {
			loading.value = false;
		}
	};

	const loadFigures = async () => {
		loading.value = true;
		error.value = null;
		try {
			const data = await import("@/pages/History/data/figures.ts");
			figures.value = data.default || [];
		} catch (err) {
			error.value = "加载人物数据失败";
			console.error("Failed to load figures:", err);
		} finally {
			loading.value = false;
		}
	};

	return {
		dynasties,
		events,
		figures,
		heritage,
		museums,
		loading,
		error,
		loadDynasties,
		loadEvents,
		loadFigures,
	};
});
```

##### 6.4 创建应用状态 Store

**文件**: `src/stores/app.ts`

```typescript
import { defineStore } from "pinia";
import { ref } from "vue";

export const useAppStore = defineStore("app", () => {
	const currentPage = ref<string>("");
	const loading = ref(false);
	const theme = ref<"light" | "dark">("light");

	const setCurrentPage = (page: string) => {
		currentPage.value = page;
	};

	const setLoading = (isLoading: boolean) => {
		loading.value = isLoading;
	};

	const toggleTheme = () => {
		theme.value = theme.value === "light" ? "dark" : "light";
	};

	return {
		currentPage,
		loading,
		theme,
		setCurrentPage,
		setLoading,
		toggleTheme,
	};
});
```
